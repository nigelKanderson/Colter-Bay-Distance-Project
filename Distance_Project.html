# Reclassify out-of-region misidentifications onto the correct regional species,
# INCLUDING inside compound labels: Labo (Eastern red bat) -> Mylu and Myse
# (N. long-eared) -> Myev. Token-wise replacement; duplicate candidates
# that result are collapsed while preserving vote order, so e.g. "Mylu/Labo" ->
# "Mylu" (becomes a single-species call) and "Lano/Myse/Mylu" -> "Lano/Myev/Mylu".
.fix_aliases <- function(x) {
  vapply(stringr::str_split(x, "/"), function(toks) {
    toks <- dplyr::recode(toks, Labo = "Mylu", Myse = "Myev")
    paste(toks[!duplicated(toks)], collapse = "/")
  }, character(1))
}

Guild-cluster models

Rather than forcing ambiguous compound calls onto a single species, group every detection into the acoustic guild by call frequency: low (Epfu / Lano / Laci), medium (Myev), and high (Mylu / Myvo / Myci / Myca / Myyu). The medium-frequency guild is a single species, so it is labeled Myev. A detection joins a guild only if every candidate species in its label belongs to that guild. data_env holds only within-guild labels and the three guilds sum to the overall total. This models activity at the level SonoBat can actually resolve and it separates guilds that respond to distance differently (low-frequency activity concentrates near the lit parking lot, while the medium- and high-frequency guilds are more active farther out), which a pooled total would cancel out.

# `assign_cluster` and the guild sets (lowfreq_spp / medfreq_spp / highfreq_spp)
# are defined in the data pipeline above, where data_env was already restricted
# to within-guild labels -- so every row here assigns to a guild.
cl_det <- data_env %>%
  mutate(cluster = assign_cluster(species)) %>%
  group_by(site, date, cluster) %>%
  summarise(detections = sum(detections), .groups = "drop")

# zero-filled: every sampled site-night x guild, carrying site-night covariates
cluster_night <- site_night %>%
  select(site, date, color, intensity, dist_km, mean_phase,
         pct_nonforest, brightness_dark, jd_c) %>%
  tidyr::crossing(cluster = c("LowFreq", "Myev", "HighFreq")) %>%
  left_join(cl_det, by = c("site", "date", "cluster")) %>%
  mutate(detections = tidyr::replace_na(detections, 0),
         site = factor(site),
         cluster = factor(cluster, levels = c("LowFreq", "Myev", "HighFreq")))

# guild x color x intensity, plus a guild-specific distance slope
cluster_glmm <- glmmTMB(
  detections ~ color * intensity + cluster * dist_km + color +
    mean_phase + pct_nonforest + brightness_dark + jd_c + I(jd_c^2) +
    (1 | site),
  data      = cluster_night,
  ziformula = ~1,
  family    = nbinom2()
)

Species models

Community-average species model (hierarchical)

Total detections are ~68% Mylu, so a pooled total mostly reflects one species. This model instead keeps one row per site-night × focal species (with zeros filled in for species not detected that night) and enters species as a random effect, so the fixed effects estimate the community-average light response rather than a Mylu-weighted total. The zero-filled cells (~39%) make the zero-inflated NB appropriate.

library(glmmTMB)

Per-species intensity × color effects

Each species gets its own intensity response, with a common color effect. Likelihood-ratio tests show species differ strongly in their intensity response (species × intensity, p < 1e-16) but not in their color response (species × color, p = 0.43), and the full species × color × intensity three-way is not worth its parameters (AIC and BIC both prefer dropping it). So we use species * intensity + color * intensity — supported, better-fitting, and interpretable without a three-way. The red-vs-white gap is therefore consistent across species; the panels differ in their intensity shapes.

make_taxon_distance <- function(level, out_tag, pretty,
                                exclude_rx = NULL, n_top = 9, min_sitenights = 30,
                                drop_sites = NULL) {
  sfx <- if (is.null(drop_sites)) "" else "_nofar"
  sub_note <- if (is.null(drop_sites))
    "Model-estimated marginal means ± 95% CI (negative-binomial GLMM)" else
    "Far sites (GRTE13/14/15) removed — marginal means ± 95% CI"
  sub_note_log <- if (is.null(drop_sites))
    "Model-estimated marginal means (log scale)" else
    "Far sites (GRTE13/14/15) removed — marginal means (log scale)"
  det <- sp_tax %>% filter(!is.na(.data[[level]])) %>% rename(taxon = all_of(level))
  if (!is.null(exclude_rx)) det <- det %>% filter(!str_detect(taxon, exclude_rx))
  keep <- det %>% group_by(taxon) %>%
    summarise(d = n(), sn = n_distinct(paste(site, date)), .groups = "drop") %>%
    filter(sn >= min_sitenights) %>% slice_max(d, n = n_top) %>% arrange(desc(d)) %>%
    pull(taxon)

  det_kept <- det %>% filter(taxon %in% keep) %>%
    group_by(site, date, color, intensity, taxon) %>%
    summarise(detections = n(), .groups = "drop")

  grid <- event_covariates %>%
    mutate(jd_c = jd - mean(jd, na.rm = TRUE)) %>%
    tidyr::crossing(taxon = keep) %>%
    left_join(det_kept, by = c("site","date","color","intensity","taxon")) %>%
    mutate(detections = tidyr::replace_na(detections, 0),
           taxon = factor(taxon, levels = keep), site = factor(site),
           color = factor(color),
           intensity = factor(intensity, levels = c("10","30","50","70","100")))
  if (!is.null(drop_sites))
    grid <- grid %>% filter(!site %in% drop_sites) %>% droplevels()

  fit <- function(extra) glmmTMB(
    as.formula(paste("detections ~ taxon * dist_km + color + intensity + mean_phase +",
                     extra, "jd_c + I(jd_c^2) + (1 | site)")),
    data = grid, ziformula = ~1, family = nbinom2())

Significant-taxa distance hierarchy

Synthesises the per-taxon distance results into one taxonomic tree. At each level (family → subfamily → genus → species) we test every taxon with ≥30 site-nights for a distance slope (joint taxon × dist_km NB-GLMM), keep those with p < 0.05, and draw each as a miniature of its model-predicted detections-vs-distance curve (±95% CI, auto-scaled within its box). Nodes are colored by direction (more active away from vs. toward the lot) and linked to their nearest significant ancestor. p is uncorrected across the many taxa, so single-star nodes should be read cautiously.

.first_by <- function(df, key, cols) df %>% group_by(across(all_of(key))) %>%
  slice_max(nn, n = 1, with_ties = FALSE) %>% ungroup() %>% select(all_of(c(key, cols)))
.sub2 <- .first_by(.tax %>% filter(!is.na(Subfamily)) %>% count(Subfamily, Family, wt = nn, name = "nn"), "Subfamily", "Family")
.gen2 <- .first_by(.tax %>% filter(!is.na(Genus)) %>% count(Genus, Subfamily, Family, wt = nn, name = "nn"), "Genus", c("Subfamily", "Family"))
.sp2  <- .first_by(.tax %>% filter(!is.na(Species_bin)) %>% count(Species_bin, Genus, Subfamily, Family, wt = nn, name = "nn"), "Species_bin", c("Genus", "Subfamily", "Family"))
.stbl <- function(m) !is.null(m) && isTRUE(m$sdr$pdHess) && !any(is.na(suppressWarnings(sqrt(diag(vcov(m)$cond)))))
.pstar <- function(p) ifelse(is.na(p), "", ifelse(p < .001, "***", ifelse(p < .01, "**", ifelse(p < .05, "*", ""))))
INT_LV <- c("10", "30", "50", "70", "100")

# per-level distance model -> per-taxon slope/p and predicted distance curve
hier_level <- function(level, drop_sites = NULL, exclude_rx = NULL, min_sn = 30) {
  det <- sp_tax %>% filter(!is.na(.data[[level]])) %>% rename(taxon = all_of(level))
  if (!is.null(exclude_rx)) det <- det %>% filter(!str_detect(taxon, exclude_rx))
  keep <- det %>% group_by(taxon) %>% summarise(sn = n_distinct(paste(site, date)), .groups = "drop") %>%
    filter(sn >= min_sn) %>% pull(taxon)
  if (length(keep) < 1) return(NULL)
  det_kept <- det %>% filter(taxon %in% keep) %>%
    group_by(site, date, color, intensity, taxon) %>% summarise(detections = n(), .groups = "drop")
  grid <- event_covariates %>% mutate(jd_c = jd - mean(jd, na.rm = TRUE)) %>%
    tidyr::crossing(taxon = keep) %>%
    left_join(det_kept, by = c("site", "date", "color", "intensity", "taxon")) %>%
    mutate(detections = tidyr::replace_na(detections, 0), taxon = factor(taxon, levels = keep),
           site = factor(site), color = factor(color), intensity = factor(intensity, levels = INT_LV))
  if (!is.null(drop_sites)) grid <- grid %>% filter(!site %in% drop_sites) %>% droplevels()
  cand <- c("mean_phase + pct_nonforest + brightness_dark + jd_c + I(jd_c^2)",
            "mean_phase + brightness_dark + jd_c + I(jd_c^2)", "mean_phase + jd_c + I(jd_c^2)",
            "mean_phase + jd_c", "jd_c")
  m <- NULL
  for (rhs in cand) {
    mi <- tryCatch(suppressWarnings(glmmTMB(
      as.formula(paste("detections ~ taxon * dist_km + color + intensity +", rhs, "+ (1 | site)")),
      data = grid, ziformula = ~1, family = nbinom2())), error = function(e) NULL)
    if (is.null(m) && !is.null(mi)) m <- mi
    if (.stbl(mi)) { m <- mi; break }
  }
  if (is.null(m)) return(NULL)
  s <- as.data.frame(emtrends(m, ~ taxon, var = "dist_km"))
  sumry <- tibble(taxon = as.character(s$taxon), est = s$dist_km.trend,
                  p = 2 * pnorm(-abs(s$dist_km.trend / s$SE)))
  dg <- seq(min(grid$dist_km), max(grid$dist_km), length.out = 40)
  cv <- as.data.frame(emmeans(m, ~ taxon | dist_km, at = list(dist_km = dg), type = "response")) %>%
    transmute(taxon = as.character(taxon), dist_km, response, lower = asymp.LCL, upper = asymp.UCL)
  list(sumry = sumry, curves = cv)
}

hier_nodes <- function(drop_sites = NULL) {
  lv <- list(Family = hier_level("Family", drop_sites), Subfamily = hier_level("Subfamily", drop_sites),
             Genus = hier_level("Genus", drop_sites, exclude_rx = "(inae|idae)$"),
             Species_bin = hier_level("Species_bin", drop_sites))
  yy <- c(Family = 4, Subfamily = 3, Genus = 2, Species_bin = 1)
  nd <- bind_rows(lapply(names(lv), function(l) { x <- lv[[l]]; if (is.null(x)) return(NULL)
    x$sumry %>% filter(!is.na(p), p < 0.05) %>% mutate(lev = l, y = yy[l]) }))
  curves <- bind_rows(lapply(names(lv), function(l) { x <- lv[[l]]; if (is.null(x)) return(NULL)
    x$curves %>% mutate(lev = l) }))
  if (nrow(nd) == 0) return(list(nd = nd, curves = curves))
  nd$Family <- NA; nd$Subfamily <- NA; nd$Genus <- NA; nd$Species_bin <- NA
  for (i in seq_len(nrow(nd))) { t <- nd$taxon[i]; l <- nd$lev[i]
    if (l == "Family") nd$Family[i] <- t
    if (l == "Subfamily") { nd$Subfamily[i] <- t; r <- .sub2[.sub2$Subfamily == t, ]; if (nrow(r)) nd$Family[i] <- r$Family[1] }
    if (l == "Genus") { nd$Genus[i] <- t; r <- .gen2[.gen2$Genus == t, ]; if (nrow(r)) { nd$Subfamily[i] <- r$Subfamily[1]; nd$Family[i] <- r$Family[1] } }
    if (l == "Species_bin") { nd$Species_bin[i] <- t; r <- .sp2[.sp2$Species_bin == t, ]; if (nrow(r)) { nd$Genus[i] <- r$Genus[1]; nd$Subfamily[i] <- r$Subfamily[1]; nd$Family[i] <- r$Family[1] } } }
  nd$key <- paste(nd$lev, nd$taxon, sep = ":"); nd$stars <- .pstar(nd$p)
  nd$dir <- ifelse(nd$est > 0, "Away from lot", "Toward lot")
  haskey <- function(lv2, t) if (is.na(t)) NA else { k <- paste0(lv2, ":", t); if (k %in% nd$key) k else NA }
  nd$parent <- NA_character_
  for (i in seq_len(nrow(nd))) { l <- nd$lev[i]
    anc <- switch(l,
      Species_bin = c(haskey("Genus", nd$Genus[i]), haskey("Subfamily", nd$Subfamily[i]), haskey("Family", nd$Family[i])),
      Genus = c(haskey("Subfamily", nd$Subfamily[i]), haskey("Family", nd$Family[i])),
      Subfamily = c(haskey("Family", nd$Family[i])), Family = NA_character_)
    anc <- anc[!is.na(anc)]; if (length(anc)) nd$parent[i] <- anc[1] }
  list(nd = nd, curves = curves)
}

.SPX <- 1.6; .BOXW <- 1.15; .BOXH <- 0.56
.PAL_D <- c("Away from lot" = "#2C7FB8", "Toward lot" = "#D95F02")
hier_layout <- function(nd) {
  fam <- unique(na.omit(nd$Family)); sub <- unique(na.omit(nd$Subfamily))
  gen <- unique(na.omit(nd$Genus)); spp <- unique(na.omit(nd$Species_bin))
  nd$k1 <- match(nd$Family, fam); nd$k2 <- match(nd$Subfamily, sub); nd$k3 <- match(nd$Genus, gen); nd$k4 <- match(nd$Species_bin, spp)
  nd <- nd %>% mutate(across(c(k1, k2, k3, k4), ~tidyr::replace_na(., 0)))
  nd <- nd[order(nd$k1, nd$k2, nd$k3, nd$k4), ]
  ch <- split(nd$key, nd$parent); env <- new.env(); env$cnt <- 0; env$x <- setNames(rep(NA_real_, nrow(nd)), nd$key)
  assign_x <- function(k) { kids <- ch[[k]]
    if (is.null(kids)) { env$cnt <- env$cnt + 1; env$x[k] <- env$cnt }
    else { kids <- kids[order(match(kids, nd$key))]; for (c in kids) assign_x(c); env$x[k] <- mean(env$x[kids]) } }
  roots <- nd$key[is.na(nd$parent)]; roots <- roots[order(match(roots, nd$key))]; for (r in roots) assign_x(r)
  nd$x <- env$x[nd$key] * .SPX; nd
}

Bayesian distance hierarchy (brms, partial pooling)

A Bayesian counterpart to the hierarchy above. A single negative-binomial model partially pools the distance slope up the taxonomy, (1 + dist_c | Family/Subfamily/Genus/Species), so each taxon’s slope borrows strength from its relatives and sparsely-sampled taxa are regularised toward their group mean (no per-taxon multiple testing). “Significance” becomes the posterior probability of direction, P(dir) ≥ 0.95 (* ≥0.95, ** ≥0.975, *** ≥0.99). The model is fit on all species with ≥5 site-nights but, as above, only taxa with ≥30 site-nights of their own data are displayed.

Fits are cached under output/brms_fits/: the first knit fits three models (~15 min each via cmdstanr), later knits reuse the cache. Delete the cache files to force a refit. Requires brms + a working cmdstan; if unavailable the section is skipped.

  requireNamespace("cmdstanr", quietly = TRUE) &&
  !is.null(tryCatch(cmdstanr::cmdstan_version(), error = function(e) NULL))

if (!.brms_ok) {
  message("brms/cmdstan not available, skipping Bayesian hierarchy. Install brms + cmdstan to enable.")
} else {
  suppressMessages({ library(brms); library(posterior) })
  dir.create("output/brms_fits", showWarnings = FALSE, recursive = TRUE)

  # species-level zero-filled grid (>=5 site-nights) with full lineage + scaled covariates
  brms_build_grid <- function(drop_sites = NULL) {
    ev <- insect_events %>%
      transmute(site, date = as.character(as.Date(date)), color, intensity,
                dist_km, jd, mean_phase, pct_nonforest, brightness_dark)
    spp <- sp_tax %>%
      filter(!is.na(Species_bin), !is.na(Family), !is.na(Subfamily), !is.na(Genus)) %>%
      mutate(date = as.character(as.Date(date)))
    if (!is.null(drop_sites)) { ev <- ev %>% filter(!site %in% drop_sites)
                                spp <- spp %>% filter(!site %in% drop_sites) }
    keep <- spp %>% group_by(Species_bin) %>%
      summarise(sn = n_distinct(paste(site, date)), .groups = "drop") %>%
      filter(sn >= 5) %>% pull(Species_bin)
    lin <- spp %>% filter(Species_bin %in% keep) %>%
      distinct(Species_bin, Genus, Subfamily, Family) %>%
      group_by(Species_bin) %>% slice(1) %>% ungroup()
    det <- spp %>% filter(Species_bin %in% keep) %>%
      group_by(site, date, color, intensity, Species_bin) %>%
      summarise(detections = n(), .groups = "drop")
    grid <- ev %>% tidyr::crossing(Species_bin = keep) %>%
      left_join(det, by = c("site","date","color","intensity","Species_bin")) %>%
      mutate(detections = tidyr::replace_na(detections, 0)) %>%
      left_join(lin, by = "Species_bin") %>%
      mutate(across(c(Family, Subfamily, Genus, Species_bin, site, color), factor),
             intensity = factor(intensity, levels = c("10","30","50","70","100")),
             jd_c = jd - mean(jd, na.rm = TRUE),
             dist_c = dist_km - mean(dist_km, na.rm = TRUE),
             mphase_z = as.numeric(scale(mean_phase)),
             nonf_z   = as.numeric(scale(pct_nonforest)),
             bdark_z  = as.numeric(scale(brightness_dark)))
    grid
  }

  brms_fit_hier <- function(grid, cache) {
    form <- bf(detections ~ dist_c + color + intensity + mphase_z + nonf_z + bdark_z +
                 jd_c + I(jd_c^2) + (1 | site) +
                 (1 + dist_c | Family/Subfamily/Genus/Species_bin))
    pri <- c(prior(normal(0, 1), class = b), prior(exponential(1), class = sd),
             prior(lkj(2), class = cor))
    brm(form, data = grid, family = negbinomial(), prior = pri,
        chains = 4, iter = 2000, warmup = 1000, cores = 4, backend = "cmdstanr",
        adapt_delta = 0.9, max_treedepth = 10, seed = 1, refresh = 0, file = cache)
  }

  # extract partially-pooled slopes + posterior-predicted curves; draw the tree
  brms_hier_plot <- function(fit, grid, title, min_sn = 30, pthr = 0.95) {
    lin <- grid %>% distinct(Family, Subfamily, Genus, Species_bin) %>%
      mutate(across(everything(), as.character))
    dr <- posterior::as_draws_df(fit)
    getc <- function(nm) if (nm %in% names(dr)) dr[[nm]] else rep(0, nrow(dr))
    b0 <- dr[["b_dist_c"]]; a0 <- dr[["b_Intercept"]]
    dgc <- seq(min(grid$dist_c), max(grid$dist_c), length.out = 40)
    cumab <- function(lev, f, s, g, sp) {
      a <- a0 + getc(sprintf("r_Family[%s,Intercept]", f))
      b <- b0 + getc(sprintf("r_Family[%s,dist_c]", f))
      if (lev %in% c("Subfamily","Genus","Species_bin")) {
        a <- a + getc(sprintf("r_Family:Subfamily[%s_%s,Intercept]", f, s))
        b <- b + getc(sprintf("r_Family:Subfamily[%s_%s,dist_c]", f, s)) }
      if (lev %in% c("Genus","Species_bin")) {
        a <- a + getc(sprintf("r_Family:Subfamily:Genus[%s_%s_%s,Intercept]", f, s, g))
        b <- b + getc(sprintf("r_Family:Subfamily:Genus[%s_%s_%s,dist_c]", f, s, g)) }
      if (lev == "Species_bin") {
        a <- a + getc(sprintf("r_Family:Subfamily:Genus:Species_bin[%s_%s_%s_%s,Intercept]", f, s, g, sp))
        b <- b + getc(sprintf("r_Family:Subfamily:Genus:Species_bin[%s_%s_%s_%s,dist_c]", f, s, g, sp)) }
      list(a = a, b = b)

Moth color × intensity, by subfamily

Subfamily-resolution counterpart to the overall color × intensity model (storyboard panel 2.7). Each facet is one well-sampled subfamily; the White-vs-Red contrast is held common across subfamilies and marked at each intensity.

# Color x intensity marginal means at subfamily resolution (full + far-removed).
make_taxon_ci <- function(level, out_tag, pretty, exclude_rx = NULL,
                          n_top = 9, min_sitenights = 30, drop_sites = NULL) {
  sfx <- if (is.null(drop_sites)) "" else "_nofar"
  det <- sp_tax %>% filter(!is.na(.data[[level]])) %>% rename(taxon = all_of(level))
  if (!is.null(exclude_rx)) det <- det %>% filter(!str_detect(taxon, exclude_rx))
  keep <- det %>% group_by(taxon) %>%
    summarise(d = n(), sn = n_distinct(paste(site, date)), .groups = "drop") %>%
    filter(sn >= min_sitenights) %>% slice_max(d, n = n_top) %>% arrange(desc(d)) %>%
    pull(taxon)

  det_kept <- det %>% filter(taxon %in% keep) %>%
    group_by(site, date, color, intensity, taxon) %>%
    summarise(detections = n(), .groups = "drop")

  grid <- event_covariates %>%
    mutate(jd_c = jd - mean(jd, na.rm = TRUE)) %>%
    tidyr::crossing(taxon = keep) %>%
    left_join(det_kept, by = c("site","date","color","intensity","taxon")) %>%
    mutate(detections = tidyr::replace_na(detections, 0),
           taxon = factor(taxon, levels = keep), site = factor(site),
           color = factor(color),
           intensity = factor(intensity, levels = c("10","30","50","70","100")))
  if (!is.null(drop_sites))
    grid <- grid %>% filter(!site %in% drop_sites) %>% droplevels()

  stable <- function(m) !is.null(m) && isTRUE(m$sdr$pdHess) &&
    !any(is.na(suppressWarnings(sqrt(diag(vcov(m)$cond)))))
  # Progressively simpler mean structures until SEs are estimable. Fine
  # taxonomic resolution on the far-removed (11-site) data can over-parameterise;
  # habitat covariates and the date quadratic are the first to drop.
  cand <- c(
    "mean_phase + pct_nonforest + brightness_dark + dist_km + jd_c + I(jd_c^2)",
    "mean_phase + brightness_dark + dist_km + jd_c + I(jd_c^2)",
    "mean_phase + dist_km + jd_c + I(jd_c^2)",
    "mean_phase + dist_km + jd_c",
    "dist_km + jd_c")
  m <- NULL
  for (rhs in cand) {
    mi <- tryCatch(suppressWarnings(glmmTMB(
      as.formula(paste("detections ~ taxon * intensity + color * intensity +",
                       rhs, "+ (1 | site)")),
      data = grid, ziformula = ~1, family = nbinom2())), error = function(e) NULL)
    if (is.null(m) && !is.null(mi)) m <- mi   # keep last fitted as a floor
    if (stable(mi)) { m <- mi; break }
  }

  # White-vs-Red ratio at each intensity (common across taxa)
  cc <- as.data.frame(contrast(emmeans(m, ~ color | intensity, type = "response"),
                               method = "revpairwise")) %>%
    mutate(stars = ifelse(p.value < .001, "***",
                   ifelse(p.value < .01, "**",
                   ifelse(p.value < .05, "*", ""))),
           intensity = factor(intensity, levels = c("10","30","50","70","100")))
  cat("\n===", pretty, "White/Red ratio by intensity ===\n")
  print(cc %>% transmute(intensity, ratio = round(ratio, 2),
                         p = signif(p.value, 3), stars))

  e <- as.data.frame(emmeans(m, ~ color * intensity | taxon, type = "response"))
  names(e)[names(e) %in% c("response","rate","emmean")] <- "est"
  e <- e %>% mutate(taxon = factor(taxon, levels = keep))

  fig <- ggplot(e, aes(factor(intensity), est, color = color, group = color)) +
    geom_line(linewidth = 0.6, position = position_dodge(0.3)) +
    geom_point(size = 1.6, position = position_dodge(0.3)) +
    geom_errorbar(aes(ymin = asymp.LCL, ymax = asymp.UCL), width = 0.15,
                  position = position_dodge(0.3)) +
    geom_text(data = cc, aes(x = intensity, y = Inf, label = stars),
              inherit.aes = FALSE, vjust = 1.3, size = 3.4, color = "#111111") +
    scale_color_manual(values = c(R = col_red, W = col_white),
                       labels = c(R = "Red", W = "White"), name = "Light color") +
    scale_y_continuous(expand = expansion(mult = c(0.05, 0.15))) +
    facet_wrap(~ taxon, scales = "free_y") +
    labs(title = paste0("Moth Detections by Light Color × Intensity, by ", pretty),
         subtitle = if (is.null(drop_sites))
           "Marginal means ± 95% CI; asterisks: White vs Red (common across taxa)" else
           "Far sites (GRTE13/14/15) removed; asterisks: White vs Red (common across taxa)",
         x = "Intensity (%)", y = "Predicted detections per site-night") +
    bat_theme + theme(strip.text = element_text(face = "bold.italic", size = 11),
                      legend.position = "top")
  ggsave(sprintf("output/figures/moth_%s_colorintensity_panels%s.png", out_tag, sfx),
         fig, width = 10, height = 6.5, dpi = 300, bg = "white")
  print(fig)
}

Robustness / Sensitivity Analysis

The three far sites (~7 km) sit in a large gap beyond the rest (0–2 km), and GRTE15 in particular is a high-detection outlier. Refit the key models — the overall bat model, the guild model, and the moth activity model — with these sites removed, and compare the color and distance effects. Color effects are expected to hold; the distance/guild slopes are the ones likely to hinge on the far sites.

library(emmeans)

# White-vs-Red ratio (+ p) from any model with a `color` term
.color_wr <- function(m) {
  ct <- as.data.frame(contrast(emmeans(m, ~ color, type = "response"),
                               method = "revpairwise"))
  sprintf("%.2f (p=%.3f)", ct$ratio[1], ct$p.value[1])
}
# marginal distance slope (log scale) via emtrends; `by` gives per-guild slopes
.dist_slope <- function(m, by = NULL) {
  et <- as.data.frame(if (is.null(by)) emtrends(m, ~ 1, var = "dist_km")
                      else emtrends(m, reformulate(by), var = "dist_km"))
  if (is.null(by)) sprintf("%.2f", et$dist_km.trend[1])
  else setNames(round(et$dist_km.trend, 2), et[[by]])
}

GRTE05 sensitivity

GRTE05 sits ~116 m from the lit GRTE16 yet records ~2% of GRTE16’s bat calls on the same nights (median ratio over 24 shared nights), while its nightly totals still track GRTE16 (r ≈ 0.58). It also has the most near-empty nights (8 nights < 5 calls; no other site has more than 1), a two-week gap in early July, and an unusually narrow species mix (93% Mylu). Moth catches there are typical. This pattern points to reduced detector sensitivity or a cluttered placement rather than real bat avoidance, so the key models are refit with and without GRTE05. The second table swaps brightness_dark for brightness_white100 (sky brightness with the lot lights on at white 100%). GRTE09 and GRTE10 have no white100 photos, so both white and dark versions there are fit on the 12 sites that have them; ΔAIC compares white vs dark on identical data. Brightness effects are reported per 1 SD of the covariate (log scale).

# brightness coefficient rescaled to per-SD of the covariate in the fitted data
.bright_sd <- function(m, term) {
  ct <- summary(m)$coefficients$cond
  s  <- sd(model.frame(m)[[term]])
  sprintf("%.2f (p=%.3f)", ct[term, "Estimate"] * s, ct[term, "Pr(>|z|)"])
}
